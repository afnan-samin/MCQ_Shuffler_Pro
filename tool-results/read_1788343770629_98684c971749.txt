   215→      toast({

   319→        title: "✅ রঙ-অনুযায়ী সিরিয়াল করা .docx ডাউনলোড হয়েছে",

   323→            : `${plan.size} টি প্রশ্ন রঙ-সেকশন অনুযায়ী ১ থেকে নম্বর পেয়েছে। হেডার/ইকুয়েশন/ছবি অক্ষত।`,
   324→      });
   325→    } catch (e) {
   326→      toast({ title: "সিরিয়াল করা যায়নি", description: String(e), variant: "destructive" });
   327→    } finally {
   328→      setColorSerialBusy(false);
   329→    }

   352→          ? "প্রতি সেটের সিরিয়াল ক্রম আলাদা — এক সেটের ক্রম আরেক সেটের সাথে মিলবে না।"

   376→          ? "প্রতি সেট আলাদা পেজে, সিরিয়াল ১,২,৩… করা। ফরম্যাট হুবহু অক্ষত।"

   383→    }

   390→      await downloadSerialFixedDocx({
   391→        originalFile: docx.file,
   392→        xml: docx.xml,
   393→        questions: docx.parse.questions,
   394→        baseName: docx.baseName,
   395→      });
   396→      toast({
   397→        title: "🔧 সিরিয়াল ঠিক করা .docx ডাউনলোড হয়েছে",
   398→        description: "অরিজিনাল অর্ডারেই প্রশ্নগুলো, সিরিয়াল ১..N বসানো — ফরম্যাট হুবহু অক্ষত।",
   399→      });
   400→    } catch (e) {
   401→      toast({ title: "ডাউনলোডে সমস্যা", description: String(e), variant: "destructive" });
   402→    } finally {
   403→      setFixing(false);
   404→    }
   405→  };
   406→
   407→  const handleDocxCopySet = async (si: number) => {

   425→
   426→  // শব্দ-ধরে এনকোডিং ডিটেক্টর (ডিটেক্ট হলেই চলে)
   427→  const encData = useMemo(() => {
   428→    if (docx?.parse) {
   429→      const s = analyzeText(docx.parse.fullText);
   430→      return { stats: s, dominant: s.dominant };
   431→    }
   432→    if (parsed) {
   433→      const s = analyzeText(rawText);
   434→      return { stats: s, dominant: s.dominant };
   435→    }
   436→    return { stats: null as EncodingStats | null, dominant: null as Enc | null };
   437→  }, [docx, parsed, rawText]);
   438→
   439→  const selectAll = () => {
   440→    const questions = docx?.parse ? docx.parse.questions : parsed?.questions ?? [];
   441→    setSelected(new Set(questions.map((q) => q.id)));
   442→  };
   443→
   444→  const selectNone = () => setSelected(new Set());
   445→
   446→  const selectRange = (fromPos: number, toPos: number) => {
   447→    const questions = docx?.parse ? docx.parse.questions : parsed?.questions ?? [];
   448→    const next = new Set<number>();
   449→    for (let i = fromPos; i <= toPos && i < questions.length; i++) next.add(questions[i].id);
   450→    setSelected(next);
   451→  };
   452→
   453→  // ---- শাফল গেট ----
   454→  const activeSerial = docx?.parse ? docx.parse.serial : parsed?.serial ?? null;
   455→  const activeCount = docx?.parse ? docx.parse.questions.length : parsed?.questions.length ?? 0;
   456→  const serialOk = activeSerial?.status === "ok";
   457→
   458→  const gateReason = useMemo(() => {
   459→    if (!docx && !parsed) return "প্রথমে প্রশ্ন ডিটেক্ট করুন";
   460→    if (activeCount === 0) return "কোনো প্রশ্ন পাওয়া যায়নি";
   461→    // docx মোডে সিরিয়াল gate নেই — serial replace ON থাকলে পজিশন-ভিত্তিক নম্বরেই সব ঠিক হয়ে যায়
   462→    if (!docx && !serialOk && !allowBroken) return "সিরিয়াল ঠিক নেই — অটো নম্বরিং ঠিক করুন অথবা 'যেভাবে আছে তেভাবে চালান' চালু করুন";
   463→    if (selected.size === 0) return "অন্তত একটি প্রশ্ন সিলেক্ট করুন";
   464→    if (selected.size < 2) return "অন্তত ২ টি প্রশ্ন সিলেক্ট করুন";
   465→    return null;
   466→  }, [docx, parsed, activeCount, serialOk, allowBroken, selected.size]);
   467→
   468→  const canShuffle = gateReason === null;
   469→
   470→  // ---- TEXT mode শাফল ----
   471→  const handleTextShuffle = () => {
   472→    if (!parsed || !canShuffle) return;
   473→    setShuffling(true);
   474→    try {
   475→      const pool = parsed.questions.filter((q) => selected.has(q.id));
   476→      const isOriginal = distribution === "original";
   477→      const result = buildSets(pool, {
   478→        setCount,
   479→        distribution,
   480→        shuffleWithin: isOriginal ? true : shuffleWithin,
   481→      });
   482→      setSets(result);
   483→      setSortedFlags(result.map(() => false));
   484→      setTimeout(() => {
   485→        resultsRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   486→      }, 80);
   487→      toast({
   488→        title: isOriginal
   489→          ? `🔀 ${setCount} টি সেট তৈরি — প্রতিটিতে সব ${pool.length} টি প্রশ্ন!`
   490→          : `🔀 ${setCount} টি সেট তৈরি হয়েছে!`,
   491→        description: isOriginal
   492→          ? "প্রতি সেটের সিরিয়াল ক্রম আলাদা — এক সেটের ক্রম আরেক সেটের সাথে মিলবে না।"
   493→          : `${pool.length} টি প্রশ্ন ভাগ হয়েছে। এখন Word ফাইল ডাউনলোড করতে পারেন।`,
   494→      });
   495→    } finally {
   496→      setShuffling(false);
   497→    }
   498→  };
   499→
   500→  const handleShuffle = () => {
   501→    if (docx) handleDocxShuffle();
   502→    else handleTextShuffle();
   503→  };
   504→
   505→  // ---- TEXT mode সেট অ্যাকশন ----
   506→  const toggleSort = (si: number) => {
   507→    if (!sets) return;
   508→    const wasSorted = sortedFlags[si];
   509→    setSets((prev) => {
   510→      if (!prev) return prev;
   511→      const copy = prev.map((s) => s.slice());
   512→      copy[si] = wasSorted ? shuffled(copy[si]) : copy[si].slice().sort((a, b) => a.id - b.id);
   513→      return copy;
   514→    });
   515→    setSortedFlags((prev) => {
   516→      const copy = prev.slice();
   517→      copy[si] = !wasSorted;
   518→      return copy;
   519→    });
   520→  };
   521→
   522→  // ---- TEXT mode এক্সপোর্ট ----
   523→  const handleExportDocx = async () => {
   524→    if (!sets) return;
   525→    setBusy("docx");
   526→    try {
   527→      await exportDocx(sets, exportOpts);
   528→      toast({ title: "✅ Word ফাইল ডাউনলোড হয়েছে", description: "প্রতিটি সেট আলাদা পেজে আছে।" });
   529→    } catch (e) {
   530→      toast({ title: "ডাউনলোডে সমস্যা", description: String(e), variant: "destructive" });
   531→    } finally {
   532→      setBusy(null);
   533→    }
   534→  };
   535→
   536→  const handleExportDoc = () => {
   537→    if (!sets) return;
   538→    try {
   539→      exportDocHtml(sets, exportOpts);
   540→      toast({ title: "✅ .doc ফাইল ডাউনলোড হয়েছে" });
   541→    } catch (e) {
   542→      toast({ title: "ডাউনলোডে সমস্যা", description: String(e), variant: "destructive" });
   543→    }
   544→  };
   545→
   546→  const handlePrint = () => {
   547→    if (!sets) return;
   548→    try {
   549→      printSets(sets, exportOpts);
   550→    } catch (e) {
   551→      toast({ title: "প্রিন্ট খোলা যায়নি", description: String(e), variant: "destructive" });
   552→    }
   553→  };
   554→
   555→  const handleCopySet = async (si: number) => {
   556→    if (!sets) return;
   557→    try {
   558→      const { setClipboardText } = await import("@/lib/mcq/exporter");
   559→      await copyToClipboard(setClipboardText(si, sets[si], exportOpts));
   560→      setCopiedSet(si);
   561→      setTimeout(() => setCopiedSet(null), 1800);
   562→    } catch (e) {
   563→      toast({ title: "কপি করা যায়নি", description: String(e), variant: "destructive" });
   564→    }
   565→  };
   566→
   567→  const handleCopyAll = async () => {
   568→    if (!sets) return;
   569→    setBusy("copyall");
   570→    try {
   571→      await copyToClipboard(allSetsClipboardText(sets, exportOpts));
   572→      toast({ title: "📋 সব সেট কপি হয়েছে", description: "Word-এ পেস্ট করলেই পাবেন। পেজ ব্রেক চাইলে .docx ডাউনলোড করুন।" });
   573→    } catch (e) {
   574→      toast({ title: "কপি করা যায়নি", description: String(e), variant: "destructive" });
   575→    } finally {
   576→      setBusy(null);
   577→    }
   578→  };
   579→
   580→  return (
   581→    <div className="min-h-screen flex flex-col bg-gradient-to-b from-emerald-50/60 via-background to-background">
   582→      {/* হেডার */}
   583→      <header className="border-b bg-white/80 backdrop-blur dark:bg-background/80">
   584→        <div className="mx-auto flex max-w-5xl flex-wrap items-center gap-3 px-4 py-4">
   585→          <div className="flex h-11 w-11 items-center justify-center rounded-xl bg-emerald-600 text-white shadow-sm">
   586→            <Dices className="h-6 w-6" />
   587→          </div>
   588→          <div className="min-w-0 flex-1">
   589→            <h1 className="text-xl font-bold tracking-tight md:text-2xl">MCQ Shuffler Pro</h1>
   590→            <p className="text-xs text-muted-foreground md:text-sm">
   591→              শাফল • সেট তৈরি • .docx ফরম্যাট হুবহু প্রিজার্ভ (ট্যাব, ইকুয়েশন, Bijoy) • শব্দ-ধরে ডিটেক্টর
   592→            </p>
   593→          </div>
   594→          <div className="flex flex-wrap items-center gap-2">
   595→            <Badge className="gap-1 bg-emerald-100 text-emerald-800 hover:bg-emerald-100">
   596→              <Zap className="h-3 w-3" /> ১০০% ফ্রি
   597→            </Badge>
   598→            <Badge variant="secondary" className="gap-1">
   599→              <ShieldCheck className="h-3 w-3" /> ডেটা ব্রাউজারেই থাকে
   600→            </Badge>
   601→          </div>
   602→        </div>
   603→      </header>
   604→
   605→      {/* মেইন */}
   606→      <main className="mx-auto w-full max-w-5xl flex-1 space-y-5 px-3 py-6 sm:px-4">
   607→        <InputCard
   608→          rawText={rawText}
   609→          onTextChange={handleTextChange}
   610→          onDetect={handleDetect}
   611→          onSample={handleSample}
   612→          onDocxFile={handleDocxFile}
   613→          onTextFileLoaded={loadAndDetect}
   614→          detecting={detecting}
   615→          detected={parsed !== null && parsed.questions.length > 0}
   616→          docxLoading={docxLoading}
   617→        />
   618→
   619→        {docx ? (
   620→          <>
   621→            {docx.parse && (
   622→              <DocxDetectCard
   623→                parse={docx.parse}
   624→                fileName={docx.file.name}
   625→                selected={selected}
   626→                onToggle={toggleQuestion}
   627→                onSelectAll={selectAll}
   628→                onSelectNone={selectNone}
   629→                onSelectRange={selectRange}
   630→                onSerialFix={handleDocxSerialFix}
   631→                fixing={fixing}
   632→                allowBroken={allowBroken}
   633→                onAllowBrokenChange={setAllowBroken}
   634→                encStats={encData.stats}
   635→                dominant={encData.dominant}
   636→              />
   637→            )}
   638→
   639→            {colorMode && colorAnalysis ? (
   640→              <ColorSerialCard
   641→                analysis={colorAnalysis}
   642→                fileName={docx.file.name}
   643→                busy={colorSerialBusy}
   644→                onSerial={handleColorSerial}
   645→              />
   646→            ) : (
   647→              <>
   648→                <ShuffleCard
   649→                  enabled={canShuffle}
   650→                  lockReason={gateReason ?? null}
   651→                  selectedCount={selected.size}
   652→                  setCount={setCount}
   653→                  onSetCountChange={setSetCount}
   654→                  distribution={distribution}
   655→                  onDistributionChange={setDistribution}
   656→                  shuffleWithin={shuffleWithin}
   657→                  onShuffleWithinChange={setShuffleWithin}
   658→                  onShuffle={handleShuffle}
   659→                  shuffling={shuffling}
   660→                />
   661→
   662→                <div ref={resultsRef} className="scroll-mt-4">
   663→                  {setsDocx && docx.parse && (
   664→                    <DocxSetsResult
   665→                      sets={setsDocx}
   666→                      questions={docx.parse.questions}
   667→                      renumber={renumber}
   668→                      onRenumberChange={toggleSerialByClick}
   669→                      busy={busy}
   670→                      copiedSet={copiedSet}
   671→                      onDownload={handleDocxDownload}
   672→                      onCopySet={handleDocxCopySet}
   673→                      dominant={encData.dominant}
   674→                    />
   675→                  )}
   676→                </div>
   677→              </>
   678→            )}
   679→          </>
   680→        ) : (
   681→          <>
   682→            <DetectCard
   683→              parsed={parsed}
   684→              selected={selected}
   685→              onToggle={toggleQuestion}
   686→              onSelectAll={selectAll}
   687→              onSelectNone={selectNone}
   688→              onSelectRange={selectRange}
   689→              onAutoFix={handleAutoFix}
   690→              allowBroken={allowBroken}
   691→              onAllowBrokenChange={setAllowBroken}
   692→              fixing={fixing}
   693→              encStats={encData.stats}
   694→              dominant={encData.dominant}
   695→            />
   696→
   697→            <ShuffleCard
   698→              enabled={canShuffle}
   699→              lockReason={parsed ? gateReason : "প্রথমে প্রশ্ন ডিটেক্ট করুন"}
   700→              selectedCount={selected.size}
   701→              setCount={setCount}
   702→              onSetCountChange={setSetCount}
   703→              distribution={distribution}
   704→              onDistributionChange={setDistribution}
   705→              shuffleWithin={shuffleWithin}
   706→              onShuffleWithinChange={setShuffleWithin}
   707→              onShuffle={handleShuffle}
   708→              shuffling={shuffling}
   709→            />
   710→
   711→            <div ref={resultsRef} className="scroll-mt-4">
   712→              {sets && (
   713→                <SetsResult
   714→                  sets={sets}
   715→                  sortedFlags={sortedFlags}
   716→                  onToggleSort={toggleSort}
   717→                  onReshuffleSet={toggleSort}
   718→                  exportOpts={exportOpts}
   719→                  onExportOptsChange={setExportOpts}
   720→                  onExportDocx={handleExportDocx}
   721→                  onExportDoc={handleExportDoc}
   722→                  onPrint={handlePrint}
   723→                  onCopySet={handleCopySet}
   724→                  onCopyAll={handleCopyAll}
   725→                  busy={busy}
   726→                  copiedSet={copiedSet}
   727→                  dominant={encData.dominant}
   728→                />
   729→              )}
   730→            </div>
   731→          </>
   732→        )}
   733→      </main>
   734→
   735→      {/* ফুটার */}
   736→      <footer className="mt-auto border-t bg-white/60 py-4 dark:bg-background/60">
   737→        <div className="mx-auto max-w-5xl px-4 text-center text-xs text-muted-foreground">
   738→          MCQ Shuffler Pro — অফিস, স্কুল, কোচিং সেন্টার ও ভার্সিটির জন্য ফ্রি টুল। সব প্রসেসিং আপনার ব্রাউজারেই হয়, কোনো প্রশ্ন সার্ভারে যায় না।
   739→        </div>
   740→      </footer>
   741→    </div>
   742→  );
   743→}
