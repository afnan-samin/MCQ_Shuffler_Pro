    77→function attrStr(tag: string, name: string): string {
    78→  let m = new RegExp(`\\b${name}="([^"]*)"`).exec(tag);
    79→  if (m) return m[1];
    80→  m = new RegExp(`\\b${name}='([^']*)'`).exec(tag);
    81→  return m ? m[1] : "";
    82→}

    84→function decodeXmlEntities(s: string): string {
    85→  if (!s.includes("&")) return s;
    86→  return s.replace(/&(#[xX][0-9A-Fa-f]+|#[0-9]+|amp|lt|gt|quot|apos);/g, (_, e: string) => {
    87→    if (e === "amp") return "&";
    88→    if (e === "lt") return "<";
    89→    if (e === "gt") return ">";
    90→    if (e === "quot") return '"';
    91→    if (e === "apos") return "'";
    92→    if (e[0] === "#") {
    93→      const code = e[1] === "x" || e[1] === "X" ? parseInt(e.slice(2), 16) : parseInt(e.slice(1), 10);
    94→      if (!isNaN(code) && code > 0 && code <= 0x10ffff) return String.fromCodePoint(code);
    95→    }
    96→    return _;
    97→  });
    98→}

   100→/** প্যারা-সাবস্ট্রিংয়ের ভিতরের প্লেইন টেক্সট: w:t + m:t (ক্রমে), <w:tab/> = \t */
   101→const PARA_TEXT_RE = /<w:t(?=[\s>])[^>]*>([^<]*)<\/w:t>|<m:t(?=[\s>])[^>]*>([^<]*)<\/m:t>|<w:tab\/>/g;
   102→
   103→function paraTextOf(sub: string): string {
   104→  let out = "";
   105→  PARA_TEXT_RE.lastIndex = 0;
   106→  let m: RegExpExecArray | null;
   107→  while ((m = PARA_TEXT_RE.exec(sub))) {
   108→    if (m[0] === "<w:tab/>") out += "\t";
   109→    else out += decodeXmlEntities(m[1] ?? m[2] ?? "");
   110→  }
   111→  return out;
   112→}

   114→/**
   115→ * প্যারা-সাবস্ট্রিং থেকে paragraph-shading কী:
   116→ * pPr-এর সরাসরি w:shd থেকে (pPr-এর ভিতরের rPr-এর shd নয় — সেটা প্যারা-মার্কের)
   117→ *  - fill hex থাকলে সেটা — সাদা/auto হলে null
   118→ *  - themeFill থাকলে "theme:accent1:E6" স্টাইলের কী
   119→ */
   120→function paraShadingKeyOf(sub: string): string | null {
   121→  const pPrOpen = /<w:pPr(?=[\s/>])/.exec(sub);
   122→  if (!pPrOpen) return null;
   123→  const openEnd = sub.indexOf(">", pPrOpen.index);
   124→  if (openEnd < 0) return null;
   125→  if (sub[openEnd - 1] === "/") return null; // <w:pPr/> — খালি, শেডিং নেই
   126→  const closeAt = sub.indexOf("</w:pPr>", openEnd);
   127→  if (closeAt < 0) return null;
   128→  let block = sub.slice(openEnd + 1, closeAt);
   129→  // প্যারা-মার্কের rPr-এর shd বাদ (DOM-ভার্সনেও pPr-এর সরাসরি সন্তান দেখা হতো)
   130→  block = block.replace(/<w:rPr(?=[\s/>])[\s\S]*?<\/w:rPr>/g, "");
   131→  const shd = /<w:shd(?=[\s/>])[^>]*>/.exec(block);
   132→  if (!shd) return null;
   133→  const tag = shd[0];
   134→  const fill = attrStr(tag, "w:fill").toUpperCase();
   135→  if (fill && fill !== "AUTO" && fill !== "FFFFFF") return fill;
   136→  const themeFill = attrStr(tag, "w:themeFill");
   137→  if (themeFill) {
   138→    const shade = attrStr(tag, "w:themeFillShade");
   139→    const tint = attrStr(tag, "w:themeFillTint");
   140→    return `theme:${themeFill}${shade ? `:${shade}` : ""}${tint ? `:${tint}` : ""}`;
   141→  }
   142→  return null;
   143→}

   145→// ---------- body-children স্ক্যানার (top-level <w:p> রেঞ্জ) ----------
   146→
   147→export interface BodyChild {
   148→  /** ট্যাগের নাম (যেমন "w:p", "w:tbl", "w:bookmarkEnd") */
   149→  kind: string;
   150→  start: number;
   151→  end: number;
   152→  selfClosing: boolean;
   153→}
   154→
   155→const TAG_RE = /<(\/?)([A-Za-z][\w.-]*(?::[A-Za-z][\w.-]*)?)((?:"[^"]*"|'[^']*'|[^>"'])*?)(\/?)>/g;
   156→
   157→/**
   158→ * <w:body>…</w:body>-এর ভিতরের depth-1 এলিমেন্টগুলোর স্ট্রিং-রেঞ্জ।
   159→ * ইনডেক্স = body-children ইনডেক্স (DOM-এর kids[i]-এর সাথে হুবহু মেলে)।
   160→ * নেস্টেড w:p (টেবিল/টেক্সটবক্সের ভিতরের) depth গুনে বাদ পড়ে — শুধু
   161→ * টপ-লেভেল শিশুরাই আলাদা এন্ট্রি পায়।
   162→ */
   163→export function scanBodyChildren(xml: string): BodyChild[] {
   164→  const bodyOpen = xml.indexOf("<w:body");
   165→  if (bodyOpen < 0) throw new Error("document.xml-এ w:body পাওয়া যায়নি");
   166→  const openEnd = xml.indexOf(">", bodyOpen);
   167→  const bodyClose = xml.lastIndexOf("</w:body>");
   168→  if (openEnd < 0 || bodyClose < 0 || bodyClose <= openEnd) {
   169→    throw new Error("document.xml-এ w:body পাওয়া যায়নি");
   170→  }
   171→
   172→  const out: BodyChild[] = [];
   173→  TAG_RE.lastIndex = openEnd + 1;
   174→  let depth = 0;
   175→  let openStart = -1;
   176→  let openKind = "";
   177→  let m: RegExpExecArray | null;
   178→  while ((m = TAG_RE.exec(xml)) && m.index < bodyClose) {
   179→    const full = m[0];
   180→    const closing = m[1] === "/";
   181→    const name = m[2];
   182→    const selfSlash = m[4] === "/";
   183→    if (closing) {
   184→      depth--;
   185→      if (depth === 0 && openStart >= 0) {
   186→        out.push({ kind: openKind, start: openStart, end: m.index + full.length, selfClosing: false });
   187→        openStart = -1;
   188→        openKind = "";
   189→      }
   190→    } else if (selfSlash) {
   191→      if (depth === 0) out.push({ kind: name, start: m.index, end: m.index + full.length, selfClosing: true });
   192→    } else {
   193→      if (depth === 0) {
   194→        openStart = m.index;
   195→        openKind = name;
   196→      }
   197→      depth++;
   198→    }
   199→  }
   200→  return out;
   201→}

   213→export interface DetectedColor {
   214→  key: string;
   215→  name: string;
   216→  hex: string | null;
   217→  /** এই রঙের হেডার/সেকশন সংখ্যা */

   220→
   221→export interface ColorAnalysis {
   222→  paras: ColorPara[];
   223→  colors: DetectedColor[];
   224→  questionCount: number;
   225→  /** মোট shaded প্যারা (colors এর sections-এর যোগফল) */
   226→  shadedCount: number;
   227→}

   229→/** document.xml থেকে প্যারা-প্রতি রঙ + প্রশ্ন-শুরু তথ্য তোলে (string-level, DOM ছাড়া) */
   230→export function analyzeColorDocx(xml: string): ColorAnalysis {
   231→  const children = scanBodyChildren(xml);
   232→
   233→  // ধাপ ১: প্রতিটা প্যারার টেক্সট + রঙ + ট্যাব
   234→  const texts: string[] = [];
   235→  const colorKeys: (string | null)[] = [];
   236→  const hasTabs: boolean[] = [];
   237→  const paraChildIdx: number[] = []; // paras অ্যারের এন্ট্রি → children ইনডেক্স
   238→
   239→  for (let i = 0; i < children.length; i++) {
   240→    const ch = children[i];
   241→    if (ch.kind !== "w:p") continue; // tbl/bookmarkEnd/sectPr — রঙ-মোডে স্কিপ
   242→    if (ch.selfClosing) {
   243→      texts.push("");
   244→      colorKeys.push(null);
   245→      hasTabs.push(false);
   246→    } else {
   247→      const sub = xml.slice(ch.start, ch.end);
   248→      texts.push(paraTextOf(sub));
   249→      colorKeys.push(paraShadingKeyOf(sub));
   250→      hasTabs.push(sub.includes("<w:tab/>"));
   251→    }
   252→    paraChildIdx.push(i);
   253→  }
   254→
   255→  // ধাপ ২: প্রশ্ন-শুরু ডিটেকশন (রঙ-দেওয়া লাইন কখনো প্রশ্ন না)
   256→  const nextNonEmpty = (from: number): string | null => {
   257→    for (let j = from; j < texts.length; j++) {
   258→      if (texts[j].trim()) return texts[j];
   259→    }
   260→    return null;
   261→  };
   262→
   263→  const paras: ColorPara[] = [];
   264→  const sectionCount = new Map<string, number>();
   265→  let questionCount = 0;
   266→
   267→  for (let k = 0; k < texts.length; k++) {
   268→    const colorKey = colorKeys[k];
   269→    let isQuestion = false;
   270→    if (!colorKey) {
   271→      const si = detectSerialPrefix(texts[k]);
   272→      if (si && isQuestionStart(si, hasTabs[k], nextNonEmpty(k + 1))) {
   273→        isQuestion = true;
   274→        questionCount++;
   275→      }
   276→    } else {
   277→      sectionCount.set(colorKey, (sectionCount.get(colorKey) ?? 0) + 1);
   278→    }
   279→    paras.push({ idx: paraChildIdx[k], text: texts[k], colorKey, isQuestion });
   280→  }
   281→
   282→  const colors: DetectedColor[] = [...sectionCount.entries()]
   283→    .map(([key, sections]) => ({ key, name: colorKeyName(key), hex: colorKeyHex(key), sections }))
   284→    .sort((a, b) => b.sections - a.sections);
   285→
   286→  return { paras, colors, questionCount, shadedCount: [...sectionCount.values()].reduce((a, b) => a + b, 0) };
   287→}

   289→/** কী থেকে সোয়াচ-এর hex (থিম-কী হলে null → UI-তে নিরপেক্ষ রঙ) */
   290→export function colorKeyHex(key: string): string | null {
   291→  if (key.startsWith("theme:")) return null;
   292→  return key;

   295→/** কী থেকে দেখানোর নাম: প্যালেটে থাকলে কোড (B1), না থাকলে কাস্টম */
   296→export function colorKeyName(key: string): string {
   297→  if (key.startsWith("theme:")) return "কাস্টম রঙ (থিম)";
   298→  const code = paletteCodeOf(key);
   299→  return code ?? "কাস্টম রঙ";

   302→// ---------- হেডার-স্ট্রিপ (শাফল মোডের জন্য) ----------
   303→
   304→/**
   305→ * টপ-লেভেল শেডিং-দেওয়া (রঙ-হেডার) প্যারাগুলো document.xml থেকে সরিয়ে দেয়।
   306→ * শাফল মোডের নিয়ম (ইউজার): "header takle seta bad diye sobgolo ek serial e
   307→ * niye ese shuffle" — হেডার বাদ পড়লে বাকি সব প্রশ্ন একটাই পুল/সিরিয়াল হয়,
   308→ * আর শাফলের সময় হেডার কোনো প্রশ্নের সাথে জড়িয়ে এলোমেলো যায় না।
   309→ *
   310→ * • শুধু body-র depth-1 w:p — টেবিল/টেক্সটবক্সের ভিতরের শেড কখনো ধরা হয় না
   311→ * • ডিটেকশন হুবহু analyzeColorDocx-এর paraShadingKey নিয়মে (সাদা/auto বাদ)
   312→ * • string-level splice — বাকি বাইট byte-identical, বিশাল ফাইলেও মেমোরি-নিরাপদ
   313→ */
   314→export function stripShadedParasXml(xml: string): { xml: string; removed: number; texts: string[] } {
   315→  const children = scanBodyChildren(xml);
   316→  const cuts: Array<{ start: number; end: number; text: string }> = [];
   317→  for (const ch of children) {
   318→    if (ch.kind !== "w:p" || ch.selfClosing) continue;
   319→    const sub = xml.slice(ch.start, ch.end);
   320→    if (paraShadingKeyOf(sub)) cuts.push({ start: ch.start, end: ch.end, text: paraTextOf(sub).trim() });
   321→  }
   322→  if (!cuts.length) return { xml, removed: 0, texts: [] };
   323→  const parts: string[] = [];
   324→  let pos = 0;
   325→  for (const c of cuts) {
   326→    parts.push(xml.slice(pos, c.start));
   327→    pos = c.end;
   328→  }
   329→  parts.push(xml.slice(pos));
   330→  return { xml: parts.join(""), removed: cuts.length, texts: cuts.map((c) => c.text) };
   331→}

   333→// ---------- নন-MCQ লাইন ডিটেক্টর (রঙহীন হেডার/শিরোনাম) ----------
   334→
   335→/** শাফল-পাইপলাইন থেকে বাদ পড়া লাইন — কারণসহ (UI-তে আলাদা লিস্টে দেখানো হয়) */
   336→export interface BlockedLine {
   337→  /** বাদ পড়া লাইনের টেক্সট */
   338→  text: string;
   339→  /** "color" = রঙ-হেডার (শেডিং ছিল), "pattern" = রঙ নেই কিন্তু টেক্সট-প্যাটার্নে হেডার/নন-MCQ */
   340→  reason: "color" | "pattern";
   341→}
   342→
   343→/**
   344→ * রঙহীন হেডার/শিরোনাম-টাইপ লাইনের প্যাটার্ন — ইউজারের নিয়ম: "jeta mcq noi
   345→ * seta jate bad dey"। প্যাটার্ন ইচ্ছাকৃতভাবে টাইট (লাইনের শুরুতে হেডার-শব্দ),
   346→ * যেন অপশন/প্রশ্ন-বডি লাইন কখনো ধরা না পড়ে — আর যা ধরা পড়ে সব UI-র
   347→ * ব্লকড-লিস্টে দেখা যায়, তাই ভুল হলে ইউজার সাথে সাথে বুঝতে পারেন।
   348→ */
   349→const NON_MCQ_RES: RegExp[] = [
   350→  /^Aa¨vq/, // Bijoy (SutonnyMJ): "অধ্যায়" — যেমন "Aa¨vq-8", "Aa¨vq 7"
   351→  /^অধ্যায়/, // Unicode: "অধ্যায়-১"
   352→  /^chapter\s*[-–—:.]?\s*\d/i, // English: "Chapter 1", "chapter-2:"
   353→  /^type\s*[-–—:.]?\s*\d/i, // সেকশন হেডার: "Type-1", "type 2" (রঙ-নেই ভ্যারিয়েন্ট)
   354→];
   355→
   356→/**
   357→ * টেক্সট দেখে নন-MCQ হেডার/শিরোনাম লাইন কি না।
   358→ * ⚠️ শুধু এমন লাইনে ডাকতে হবে যারা প্রশ্ন-শুরু নয়, অপশন-লেড নয় —
   359→ * স্ট্রিপ-ফাংশন সেই গার্ডগুলো নিজেই দেয়। হেডার লাইন ছোট হয় (≤৮০ অক্ষর)।
   360→ */
   361→export function isNonMcqText(text: string): boolean {
   362→  const t = text.trim();
   363→  if (!t || t.length > 80) return false;
   364→  return NON_MCQ_RES.some((re) => re.test(t));
   365→}
   366→
   367→/**
   368→ * রঙ নেই এমন নন-MCQ হেডার/শিরোনাম লাইন (যেমন Agri ফাইলের রঙহীন "Aa¨vq-8")
   369→ * document.xml থেকে সরিয়ে দেয় — যাতে কোনো প্রশ্ন-ব্লকের সাথে জড়িয়ে
   370→ * শাফলে এলোমেলো না যায়।
   371→ *
   372→ * গার্ড (শুধু এগুলোই কাটা হয়):
   373→ *  • টপ-লেভেল (depth-1) w:p, টেবিল/টেক্সটবক্সের ভিতরের প্যারা স্পর্শ হয় না
   374→ *  • রঙ-দেওয়া প্যারা কখনো নয় (ওগুলো stripShadedParasXml-এর কাজ)
   375→ *  • প্রশ্ন-শুরু (সিরিয়াল + ট্যাব/অপশন-লেড/টিয়ার-৩) কখনো নয়
   376→ *  • অপশন-লেড লাইন (K./ক./\t…) কখনো নয় — অপশন প্রশ্নের অংশ
   377→ *  • খালি/শুধু-স্পেস প্যারা নয়
   378→ *
   379→ * রিটার্নে বাদ পড়া লাইনের টেক্সটগুলো — UI-র ব্লকড-লিস্টের জন্য।
   380→ * string-level splice — বাকি বাইট byte-identical, বিশাল ফাইলেও মেমোরি-নিরাপদ।
   381→ */
   382→export function stripNonMcqLinesXml(xml: string): { xml: string; removed: BlockedLine[] } {
   383→  const children = scanBodyChildren(xml);
   384→
   385→  // প্রথম পাস: w:p-গুলোর টেক্সট + ট্যাব + শেড (প্রশ্ন-শুরু ডিটেকশনের জন্য)
   386→  const paraIdx: number[] = []; // children-ইনডেক্স
   387→  const texts: string[] = [];
   388→  const hasTabs: boolean[] = [];
   389→  const shaded: boolean[] = [];
   390→  for (let i = 0; i < children.length; i++) {
   391→    const ch = children[i];
   392→    if (ch.kind !== "w:p") continue;
   393→    if (ch.selfClosing) {
   394→      texts.push("");
   395→      hasTabs.push(false);
   396→      shaded.push(false);
   397→    } else {
   398→      const sub = xml.slice(ch.start, ch.end);
   399→      texts.push(paraTextOf(sub));
   400→      hasTabs.push(sub.includes("<w:tab/>"));
   401→      shaded.push(!!paraShadingKeyOf(sub));
   402→    }
   403→    paraIdx.push(i);
   404→  }
   405→
   406→  const nextNonEmpty = (from: number): string | null => {
   407→    for (let j = from; j < texts.length; j++) {
   408→      if (texts[j].trim()) return texts[j];
   409→    }
   410→    return null;
   411→  };
   412→
   413→  // দ্বিতীয় পাস: কোনগুলো কাটা হবে
   414→  const cuts: Array<{ start: number; end: number; text: string }> = [];
   415→  for (let k = 0; k < paraIdx.length; k++) {
   416→    const t = texts[k];
   417→    if (!t.trim() || shaded[k]) continue;
   418→    const si = detectSerialPrefix(t);
   419→    if (si && isQuestionStart(si, hasTabs[k], nextNonEmpty(k + 1))) continue;
   420→    if (looksOptionLed(t)) continue;
   421→    if (!isNonMcqText(t)) continue;
   422→    const ch = children[paraIdx[k]];
   423→    cuts.push({ start: ch.start, end: ch.end, text: t.trim() });
   424→  }
   425→
   426→  if (!cuts.length) return { xml, removed: [] };
   427→  const parts: string[] = [];
   428→  let pos = 0;
   429→  for (const c of cuts) {
   430→    parts.push(xml.slice(pos, c.start));
   431→    pos = c.end;
   432→  }
   433→  parts.push(xml.slice(pos));
   434→  return { xml: parts.join(""), removed: cuts.map((c) => ({ text: c.text, reason: "pattern" as const })) };
   435→}

   437→// ---------- নম্বর-প্ল্যান (স্ট্যাক/নেস্টিং অ্যালগরিদম) ----------
   438→
   439→export type SerialScheme = { kind: "color"; key: string } | { kind: "continuous" };
   440→
   441→/**
   442→ * স্ট্যাক-অ্যালগরিদম v2 (ইউজারের নিয়ম হুবহু):
   443→ *  - রঙ-দেওয়া হেডার C এলে: স্ট্যাকে C আগে থাকলে C-সহ ওপরের সব বন্ধ করে
   444→ *    C নতুন সেকশন হিসেবে খোলা হয় (ভাই-সেকশন); না থাকলে সবচেয়ে ভিতরের
   445→ *    খোলা সেকশনের সন্তান হিসেবে খোলে
   446→ *  - সিলেক্ট করা রঙ X-এর হেডার এলে কাউন্টার ০ (নতুন সেকশন → ১ থেকে)
   447→ *  - X যদি স্ট্যাকে না থাকে (অভিভাবক/ভাই-সীমানায় বন্ধ হয়ে গেছে) →
   448→ *    নম্বর দেওয়া বন্ধ — "program oikane theme jabe"
   449→ *  - X-এর ভিতরের গভীর রঙের হেডার নম্বর থামায় না, তাদের প্রশ্নও
   450→ *    X-সেকশনের ক্রমের অংশ
   451→ *  - continuous: পুরো ফাইলে একটানা ১,২,৩…
   452→ *
   453→ * 🎓 অধ্যায়-সোদক সোয়াপ (বাগ-ফিক্স: "অধ্যায়ের রঙ বদলালে" সমস্যা):
   454→ *    বাস্তব ফাইলে একই স্তরের (যেমন অধ্যায়) হেডারের রঙ বদলায় — অধ্যায়-১ = B6,
   455→ *    অধ্যায়-২…৫ = B1। পুরনো অ্যালগরিদমে নতুন রঙ B1-কে আগের অধ্যায়ের *সন্তান*
   456→ *    ভেবে বসালে B6-সেকশন কখনো বন্ধ হতো না — B6-স্কিম পুরো ফাইলকে একটানা
   457→ *    নম্বর দিয়ে দিত (আউটপুট continuous-এর হুবহু কপি হয়ে যেত)।
   458→ *    এখন: fresh রঙ C এলে যদি — (১) স্ট্যাকের রুট R এ পর্যন্ত মাত্র ১ বার
   459→ *    এসে থাকে (তার একটাই সেকশন খোলা), (২) R-এর সরাসরি-সন্তান রঙের প্রমাণ
   460→ *    থাকে (R আসলেই একটা অভিভাবক-স্তর), (৩) C ও R-এর হেডার-ঘনত্ব একই
   461→ *    অর্ডারে (≤১০×) — তাহলে C হলো R-এর *সোদক* (নতুন অধ্যায়): রুট-সেকশন
   462→ *    বন্ধ করে C নতুন রুট হয়। ইউজারের A5/A2/A6 নেস্টিং উদাহরণে এই নিয়ম
   463→ *    কখনো ফায়ার করে না (সেখানে রুট A5 বারবার রি-অ্যারাইভ করে)।
   464→ *
   465→ * 🛡️ ২×-গার্ড: কোনো রঙের রিস্টার্ট তার স্ট্যাক-পজিশনের ওপরে থাকা ২×-এর কম ঘন
   466→ *    (তাই তুলনামূলক উঁচু-লেভেলের) সেকশনকে বন্ধ করতে পারে না — ঘন রঙের
   467→ *    রিস্টার্ট কখনো উঁচু-লেভেলের খোলা সেকশন (যেমন সিলেক্টেড X) মেরে ফেলবে না।
   468→ *
   469→ * রিটার্ন: Map<paraIdx, newNumber> — শুধু প্রশ্ন-শুরু প্যারারাই থাকে
   470→ */
   471→export function planSerialByColor(analysis: ColorAnalysis, scheme: SerialScheme): Map<number, number> {
   472→  const plan = new Map<number, number>();
   473→  const X = scheme.kind === "color" ? scheme.key : null;
   474→
   475→  // গ্লোবাল ঘনত্ব: যে রঙের হেডার তত বেশি, সে তত গভীর-লেভেলের (Type ×২৭ ⊂ অধ্যায় ×৪)
   476→  const countOf = new Map<string, number>();
   477→  for (const c of analysis.colors) countOf.set(c.key, c.sections);
   478→
   479→  const stack: string[] = [];
   480→  const seen = new Map<string, number>(); // রঙ → এ পর্যন্ত হেডার-সংখ্যা
   481→  const kids = new Map<string, Set<string>>(); // রঙ → তার সরাসরি-সন্তান রঙের প্রমাণ (fresh-পুশ এজ)
   482→  let counter = 0;
   483→  let inX = X === null; // continuous → সব প্রশ্ন নম্বর পায়
   484→
   485→  for (const para of analysis.paras) {
   486→    if (para.colorKey) {
   487→      const C = para.colorKey;
   488→      let at = -1;
   489→      for (let s = stack.length - 1; s >= 0; s--) {
   490→        if (stack[s] === C) { at = s; break; }
   491→      }
   492→      if (at >= 0) {
   493→        // ভাই-রিস্টার্ট — ২×-গার্ড
   494→        const cCount = countOf.get(C) ?? 0;
   495→        let cut = at;
   496→        for (let s = at + 1; s < stack.length; s++) {
   497→          if ((countOf.get(stack[s]) ?? 0) * 2 <= cCount) { cut = s + 1; break; }
   498→        }
   499→        stack.length = cut;
   500→      } else if (stack.length > 0) {
   501→        // fresh রঙ — অধ্যায়-সোদক সোয়াপ কি হবে?
   502→        const root = stack[0];
   503→        const seenRoot = seen.get(root) ?? 0;
   504→        const cN = countOf.get(C) ?? 0;
   505→        const rN = countOf.get(root) ?? 0;
   506→        const hi = Math.max(cN, rN);
   507→        const lo = Math.min(cN, rN);
   508→        // প্রমাণ: রুটের অন্তত একটা সন্তান-রঙ ২+ বার এসেছে — অর্থাৎ রুটের
   509→        // ভিতরের স্তরটা রিপিট হয়ে স্ট্রাকচার প্রতিষ্ঠিত (আসল ফাইলে অধ্যায়-১-এ
   510→        // Type-সেকশন বারবার রিস্টার্ট হয়েছে; প্রথম Type-এর মাঝেই যে ভিতরের
   511→        // লেভেল প্রথমবার খুলছে সেটা অধ্যায় নয় — varsity-স্টাইল গভীর স্তর)
   512→        const childEstablished = [...(kids.get(root) ?? [])].some((ch) => (seen.get(ch) ?? 0) >= 2);
   513→        // আরও প্রমাণ: C রুটের প্রতিষ্ঠিত সন্তানদের চেয়ে রেয়ার (তাই C-ও
   514→        // সন্তানদের মতো উঁচু-লেভেলের ব্যান্ডে — গভীর-লেভেলের হেডার নয়)
   515→        const childDensest = Math.max(...[...(kids.get(root) ?? [])].map((ch) => countOf.get(ch) ?? 0));
   516→        // সবচেয়ে জোরালো প্রমাণ: C রুট ছাড়া বাকি *সব খোলা* রঙের চেয়ে কঠোরভাবে
   517→        // রেয়ার — নতুন অধ্যায়ের রঙ সবসময় চলমান Type/varsity-স্তরের চেয়ে রেয়ার
   518→        // (B1=৪ < Type=২৭ < varsity=২৫৭)। ইউজারের A5/A6-নেস্টিংয়ে A6(৩) A2(৩)-এর
   519→        // চেয়ে রেয়ার নয় — তাই ভুল সোয়াপ হয় না।
   520→        let openMin = Infinity;
   521→        for (let s = 1; s < stack.length; s++) openMin = Math.min(openMin, countOf.get(stack[s]) ?? 0);
   522→        if (seenRoot === 1 && childEstablished && cN <= childDensest && cN < openMin && hi <= lo * 10) {
   523→          stack.length = 0; // হ্যাঁ — C রুটের সোদক: রুট-সেকশন বন্ধ, C নতুন রুট
   524→        }
   525→        // fresh পুশ-এজ: parent = বর্তমান টপ
   526→        const parent = stack.length ? stack[stack.length - 1] : null;
   527→        if (parent) {
   528→          if (!kids.has(parent)) kids.set(parent, new Set());
   529→          kids.get(parent)!.add(C);
   530→        }
   531→      }
   532→      seen.set(C, (seen.get(C) ?? 0) + 1);
   533→      stack.push(C);
   534→      if (X !== null) {
   535→        if (C === X) counter = 0; // নতুন X-সেকশন (হুবহু রঙে)
   536→        inX = stack.includes(X); // স্কোপিংও হুবহু রঙে
   537→      }
   538→    } else if (para.isQuestion) {
   539→      if (inX) {
   540→        counter++;
   541→        plan.set(para.idx, counter);
   542→      }
   543→    }
   544→  }
   545→  return plan;
   546→}

   561→ * একটা প্যারা-সাবস্ট্রিং-এর w:t-স্ট্রিমে decoded-স্প্যান রিপ্লেস —
   562→ * DOM-যুগের replaceSpans-এর হুবহু সেমান্টিক্স: যে সেগমেন্টে স্প্যান শুরু
   563→ * সেখানে নতুন টেক্সট বসে, মাঝের সেগমেন্ট খালি হয়, শেষ সেগমেন্টের বাকি
   564→ * অংশ অক্ষত থাকে — মাঝের রান-মার্কআপ কখনো মোছে না।
   565→ */
   566→function replaceDecodedSpan(sub: string, segs: WtSegment[], decStart: number, decEnd: number, text: string): string {
   567→  if (decEnd <= decStart) return sub;
   568→  // সেগমেন্ট prefix-অফসেট
   569→  const prefix: number[] = [];
   570→  let acc = 0;
   571→  for (const s of segs) {
   572→    prefix.push(acc);
   573→    acc += s.text.length;
   574→  }
   575→  // decoded পজিশন → (সেগমেন্ট, লোকাল decoded)
   576→  const locate = (p: number): { si: number; local: number } => {
   577→    let lo = 0;
   578→    let hi = segs.length - 1;
   579→    let si = 0;
   580→    while (lo <= hi) {
   581→      const mid = (lo + hi) >> 1;
   582→      if (prefix[mid] <= p) {
   583→        si = mid;
   584→        lo = mid + 1;
   585→      } else hi = mid - 1;
   586→    }
   587→    return { si, local: p - prefix[si] };
   588→  };
   589→  // সেগমেন্টের লোকাল decoded → raw অফসেট (entity হিসাব করে)
   590→  const localToRaw = (si: number, local: number): number => {
   591→    const s = segs[si];
   592→    const content = sub.slice(s.rawStart, s.rawEnd);
   593→    if (content.length === s.text.length) return s.rawStart + local; // entity নেই — দ্রুত পথ
   594→    let i = 0;
   595→    let dec = 0;
   596→    while (i < content.length && dec < local) {
   597→      if (content[i] === "&") {
   598→        const semi = content.indexOf(";", i);
   599→        i = semi < 0 ? i + 1 : semi + 1;
   600→        dec += 1;
   601→      } else {
   602→        i++;
   603→        dec++;
   604→      }
   605→    }
   606→    return s.rawStart + i;
   607→  };
   608→
   609→  const a = locate(decStart);
   610→  const b = locate(decEnd);
   611→  const edits: Array<{ start: number; end: number; text: string }> = [];
   612→  if (a.si === b.si) {
   613→    const ra = localToRaw(a.si, a.local);
   614→    const rb = localToRaw(b.si, b.local);
   615→    edits.push({ start: ra, end: rb, text });
   616→  } else {
   617→    // শুরুর সেগমেন্ট: লোকাল থেকে কনটেন্ট-শেষ পর্যন্ত → নতুন টেক্সট
   618→    edits.push({ start: localToRaw(a.si, a.local), end: segs[a.si].rawEnd, text });
   619→    // মাঝের সেগমেন্ট: পুরো কনটেন্ট খালি
   620→    for (let s = a.si + 1; s < b.si; s++) edits.push({ start: segs[s].rawStart, end: segs[s].rawEnd, text: "" });
   621→    // শেষ সেগমেন্ট: কনটেন্ট-শুরু থেকে লোকাল পর্যন্ত খালি
   622→    edits.push({ start: segs[b.si].rawStart, end: localToRaw(b.si, b.local), text: "" });
   623→  }
   624→  return spliceEdits(sub, edits);
   625→}
   626→
   627→/** raw এডিটগুলো (ascending, non-overlapping) এক পাসে বসায় */
   628→function spliceEdits(sub: string, edits: Array<{ start: number; end: number; text: string }>): string {
   629→  if (!edits.length) return sub;
   630→  const sorted = [...edits].sort((x, y) => x.start - y.start);
   631→  const parts: string[] = [];
   632→  let pos = 0;
   633→  for (const e of sorted) {
   634→    parts.push(sub.slice(pos, e.start), e.text);
   635→    pos = e.end;
   636→  }
   637→  parts.push(sub.slice(pos));
   638→  return parts.join("");
   639→}
   640→
   641→/**
   642→ * plan অনুযায়ী শুধু প্রশ্ন-প্যারার সিরিয়াল বদলায় (সেপারেটর "." ডট-স্টাইলে
   643→ * নরমালাইজ — "44|" → "45.")। বাকি পুরো XML হুবহু অরিজিনাল (byte-identical):
   644→ * এডিট হয় শুধু ডিজিট/সেপারেটরের raw অক্ষর। জিরো-প্যাডিং ("01.") ফাইলের
   645→ * নিজের স্টাইল অনুযায়ী ধরে রাখা হয় (১ → "01")।
   646→ */
   647→export function applyColorSerialXml(xml: string, plan: Map<number, number>, targetSep = "."): string {
   648→  if (plan.size === 0) return xml;
   649→  const children = scanBodyChildren(xml);
   650→
   651→  const parts: string[] = [];
   652→  let lastPos = 0;
   653→  const idxs = [...plan.keys()].sort((a, b) => a - b);
   654→
   655→  for (const idx of idxs) {
   656→    const ch = children[idx];
   657→    if (!ch || ch.kind !== "w:p" || ch.selfClosing) continue;
   658→    const sub = xml.slice(ch.start, ch.end);
   659→    const edited = renumberParaSub(sub, plan.get(idx)!, targetSep);
   660→    if (edited !== sub) {
   661→      parts.push(xml.slice(lastPos, ch.start));
   662→      parts.push(edited);
   663→      lastPos = ch.end;
   664→    }
   665→  }
   666→  parts.push(xml.slice(lastPos));
   667→  return parts.join("");
   668→}
   669→
   670→/** একটা প্যারা-সাবস্ট্রিংয়ের সিরিয়াল নতুন নম্বরে বদলায় (ডিজিট+সেপারেটর) */
   671→function renumberParaSub(sub: string, newNum: number, targetSep: string): string {
   672→  const segs = collectWtSegs(sub);
   673→  if (!segs.length) return sub;
   674→
   675→  const joined = segs.map((s) => s.text).join("");
   676→  const segEnds: number[] = [];
   677→  let acc = 0;
   678→  for (const s of segs) { acc += s.text.length; segEnds.push(acc); }
   679→  const spans = serialMatchSpans(joined, segEnds);
   680→  if (!spans) return sub;
   681→
   682→  // জিরো-প্যাডিং সংরক্ষণ: "01" স্টাইলের ফাইলে ১ → "01"
   683→  let digitsText = numberToDigits(newNum, spans.enc);
   684→  const origDigits = joined.slice(spans.digitsStart, spans.digitsEnd);
   685→  const zeroChar = spans.enc === "en" ? "0" : spans.enc === "bn" ? "০" : "ø";
   686→  if (origDigits.startsWith(zeroChar) && digitsText.length < origDigits.length) {
   687→    digitsText = zeroChar.repeat(origDigits.length - digitsText.length) + digitsText;
   688→  }
   689→
   690→  let out = replaceDecodedSpan(sub, segs, spans.digitsStart, spans.digitsEnd, digitsText);
   691→  if (spans.sepEnd > spans.sepStart) {
   692→    // আগের এডিটের পরে সেগমেন্ট raw-পজিশন বদলাতে পারে — আবার স্ক্যান
   693→    const segs2 = collectWtSegs(out);
   694→    const joined2 = segs2.map((s) => s.text).join("");
   695→    const segEnds2: number[] = [];
   696→    let acc2 = 0;
   697→    for (const s of segs2) { acc2 += s.text.length; segEnds2.push(acc2); }
   698→    const spans2 = serialMatchSpans(joined2, segEnds2);
   699→    if (spans2 && spans2.sepEnd > spans2.sepStart) {
   700→      out = replaceDecodedSpan(out, segs2, spans2.sepStart, spans2.sepEnd, targetSep);
   701→    }
   702→  }
   703→  return out;
   704→}
   705→
   706→function collectWtSegs(sub: string): WtSegment[] {
   707→  const segs: WtSegment[] = [];
   708→  WT_RE.lastIndex = 0;
   709→  let m: RegExpExecArray | null;
   710→  while ((m = WT_RE.exec(sub))) {
   711→    const openEnd = m.index + m[0].indexOf(">") + 1;
   712→    segs.push({ rawStart: openEnd, rawEnd: m.index + m[0].length - "</w:t>".length, text: decodeXmlEntities(m[1]) });
   713→  }
   714→  return segs;
   715→}

   717→const DOCX_MIME = "application/vnd.openxmlformats-officedocument.wordprocessingml.document";
   718→
   719→/** রঙ-সিরিয়াল করা .docx ডাউনলোড — অরিজিনাল zip-এর বাকি সব (ছবি/স্টাইল/সেটিংস) অক্ষত */
   720→export async function downloadColorSerialDocx(params: {
   721→  originalFile: Blob;
   722→  xml: string;
   723→  plan: Map<number, number>;
   724→  baseName: string;
   725→  schemeLabel: string;
   726→}): Promise<void> {
   727→  const newXml = applyColorSerialXml(params.xml, params.plan);
   728→  const src = await JSZip.loadAsync(params.originalFile);
   729→  // নতুন zip-এ নন-ডিরেক্টরি এন্ট্রিগুলো হুবহু কপি, document.xml বদলে নতুন XML —
   730→  // (JSZip নিজে ফোল্ডার-এন্ট্রি বানায়, আর remove() রিকার্সিভ — তাই সরাসরি কপি-নির্মাণ)
   731→  const others: Array<{ path: string; data: Promise<Uint8Array> }> = [];
   732→  src.forEach((path, entry) => {
   733→    if (!entry.dir && path !== "word/document.xml") others.push({ path, data: entry.async("uint8array") });
   734→  });
   735→  const zip = new JSZip();
   736→  zip.file("word/document.xml", new TextEncoder().encode(newXml));
   737→  for (const o of others) zip.file(o.path, await o.data);
   738→  const blob = await zip.generateAsync({ type: "blob", mimeType: DOCX_MIME, compression: "DEFLATE" });
   739→
   740→  const url = URL.createObjectURL(blob);
   741→  const a = document.createElement("a");
   742→  a.href = url;
   743→  a.download = `${params.baseName} (color serial - ${params.schemeLabel}).docx`;
   744→  document.body.appendChild(a);
   745→  a.click();
   746→  a.remove();
   747→  setTimeout(() => URL.revokeObjectURL(url), 4000);
   748→}
