   527→
   528→  // ---- শাফল গেট ----

   537→    if (!docx && !serialOk && !allowBroken) return "সিরিয়াল ঠিক নেই — অটো নম্বরিং ঠিক করুন অথবা 'যেভাবে আছে তেভাবে চালান' চালু করুন";

   567→          ? "প্রতি সেটের সিরিয়াল ক্রম আলাদা — এক সেটের ক্রম আরেক সেটের সাথে মিলবে না।"
   568→          : `${pool.length} টি প্রশ্ন ভাগ হয়েছে। এখন Word ফাইল ডাউনলোড করতে পারেন।`,

   652→    }

   658→      <header className="border-b bg-white/80 backdrop-blur dark:bg-background/80">
   659→        <div className="mx-auto flex max-w-5xl flex-wrap items-center gap-3 px-4 py-4">
   660→          <div className="flex h-11 w-11 items-center justify-center rounded-xl bg-emerald-600 text-white shadow-sm">
   661→            <Dices className="h-6 w-6" />
   662→          </div>
   663→          <div className="min-w-0 flex-1">
   664→            <h1 className="text-xl font-bold tracking-tight md:text-2xl">MCQ Shuffler Pro</h1>
   665→            <p className="text-xs text-muted-foreground md:text-sm">
   666→              দুই আলাদা মোড — 🔀 শাফল+সেট তৈরি ও 🔢 রঙ-অনুযায়ী সিরিয়াল • .docx ফরম্যাট হুবহু প্রিজার্ভ (ট্যাব, ইকুয়েশন, Bijoy)

   671→              <Zap className="h-3 w-3" /> ১০০% ফ্রি

   683→        <ModeTabs mode={mode} onChange={changeMode} />

   685→        {mode === "serial" ? (

   710→          <>
   711→            <InputCard

   714→              onDetect={handleDetect}
   715→              onSample={handleSample}
   716→              onDocxFile={handleDocxFile}
   717→              onTextFileLoaded={loadAndDetect}
   718→              detecting={detecting}
   719→              detected={parsed !== null && parsed.questions.length > 0}

   722→
   723→            {docx ? (

   754→                    setCount={setCount}

   756→                    distribution={distribution}
   757→                    onDistributionChange={setDistribution}
   758→                    shuffleWithin={shuffleWithin}
   759→                    onShuffleWithinChange={setShuffleWithin}
   760→                    onShuffle={handleShuffle}
   761→                    shuffling={shuffling}
   762→                  />
   763→
   764→                  <div ref={resultsRef} className="scroll-mt-4">
   765→                    {setsDocx && docx.parse && (
   766→                      <DocxSetsResult
   767→                        sets={setsDocx}
   768→                        questions={docx.parse.questions}
   769→                        renumber={renumber}
   770→                        onRenumberChange={toggleSerialByClick}
   771→                        busy={busy}
   772→                        copiedSet={copiedSet}
   773→                        onDownload={handleDocxDownload}
   774→                        onCopySet={handleDocxCopySet}
   775→                        dominant={encData.dominant}

   778→                  </div>
   779→                </>
   780→              )
   781→            ) : (
   782→              <>
   783→                <DetectCard
   784→                  parsed={parsed}
   785→                  selected={selected}
   786→                  onToggle={toggleQuestion}
   787→                  onSelectAll={selectAll}
   788→                  onSelectNone={selectNone}
   789→                  onSelectRange={selectRange}
   790→                  onAutoFix={handleAutoFix}
   791→                  allowBroken={allowBroken}
   792→                  onAllowBrokenChange={setAllowBroken}

   801→                  selectedCount={selected.size}
   802→                  setCount={setCount}
   803→                  onSetCountChange={setSetCount}
   804→                  distribution={distribution}

   806→                  shuffleWithin={shuffleWithin}
   807→                  onShuffleWithinChange={setShuffleWithin}
   808→                  onShuffle={handleShuffle}
   809→                  shuffling={shuffling}
   810→                />

   812→                <div ref={resultsRef} className="scroll-mt-4">

   815→                      sets={sets}
   816→                      sortedFlags={sortedFlags}
   817→                      onToggleSort={toggleSort}
   818→                      onReshuffleSet={toggleSort}
   819→                      exportOpts={exportOpts}
   820→                      onExportOptsChange={setExportOpts}
   821→                      onExportDocx={handleExportDocx}
   822→                      onExportDoc={handleExportDoc}
   823→                      onPrint={handlePrint}
   824→                      onCopySet={handleCopySet}
   825→                      onCopyAll={handleCopyAll}
   826→                      busy={busy}
   827→                      copiedSet={copiedSet}
   828→                      dominant={encData.dominant}
   829→                    />
   830→                  )}
   831→                </div>
   832→              </>
   833→            )}
   834→          </>
   835→        )}
   836→      </main>
   837→
   838→      {/* ফুটার */}

   841→          MCQ Shuffler Pro — অফিস, স্কুল, কোচিং সেন্টার ও ভার্সিটির জন্য ফ্রি টুল। সব প্রসেসিং আপনার ব্রাউজারেই হয়, কোনো প্রশ্ন সার্ভারে যায় না।
   842→        </div>
   843→      </footer>
   844→    </div>
   845→  );
   846→}
